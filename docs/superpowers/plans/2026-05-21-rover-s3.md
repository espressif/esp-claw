# rover_s3 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build `application/rover_s3` — an AI rover application for M5Stack StickS3 with always-listening wake word ("Hi ESP"), hybrid on-device/cloud STT, Telegram, TTS responses, and a new reusable `cap_voice` capability component.

**Architecture:** New `application/rover_s3/` (PlatformIO, ESP32-S3, 8 MB PSRAM) wires existing `cap_rover`/`cap_unitv`/`cap_im_tg` with a new `cap_voice` component. `cap_voice` runs WakeNet+AFE always-on, falls back from MultiNet to Whisper API, and plays TTS responses through ES8311. Both voice and Telegram route through `claw_event_router` → `claw_core`.

**Tech Stack:** ESP-IDF 5.x via PlatformIO, ESP-SR (WakeNet9 + MultiNet6), `esp_codec_dev` for ES8311, M5Unified for board init/display/IMU, OpenAI Whisper + TTS APIs, `claw_llm_http_transport` for HTTP.

**Spec:** `docs/superpowers/specs/2026-05-21-rover-s3-design.md`

```cpp
/*
 * SPDX-FileCopyrightText: 2026 Espressif Systems (Shanghai) CO LTD
 * SPDX-License-Identifier: Apache-2.0
 */
#include <stdio.h>
#include <string.h>
#include "M5Unified.h"

extern "C" {
#include "setup_device.h"
}

static rover_s3_display_state_t s_last_state = (rover_s3_display_state_t)-1;
static char s_last_ip[20] = {0};
static int  s_last_batt = -2;
static bool s_last_charging = false;

extern "C" esp_err_t rover_s3_board_init(void)
{
    auto cfg = M5.config();
    cfg.output_power = true;
    cfg.internal_imu = true;
    cfg.internal_rtc = true;
    /* Audio managed by cap_voice, not M5Unified */
    cfg.internal_mic = false;
    cfg.internal_spk = false;
    cfg.led_brightness = 0;

    M5.begin(cfg);
    M5.Display.setRotation(1);   /* landscape; adjust 0-3 for desired orientation */
    M5.Display.setBrightness(128);
    M5.Display.fillScreen(TFT_BLACK);
    return ESP_OK;
}

extern "C" void rover_s3_board_update(void)
{
    M5.update();
}

extern "C" bool rover_s3_board_btn_a_pressed(void)
{
    return M5.BtnA.isPressed();
}

extern "C" bool rover_s3_board_btn_b_pressed(void)
{
    return M5.BtnB.isPressed();
}

extern "C" int rover_s3_board_get_battery_pct(void)
{
    int pct = M5.Power.getBatteryLevel();
    return (pct < 0) ? -1 : (pct > 100 ? 100 : pct);
}

extern "C" bool rover_s3_board_is_charging(void)
{
    return M5.Power.isCharging();
}

/* Label map — keep in sync with rover_s3_display_state_t */
static const char *STATE_LABELS[] = {
    "BOOT", "IDLE", "LISTEN", "THINK", "SPEAK", "EXEC", "OFFLINE"
};
static uint32_t STATE_COLORS[] = {
    TFT_DARKGREY, TFT_GREEN, TFT_CYAN, TFT_YELLOW,
    TFT_MAGENTA, TFT_ORANGE, TFT_RED
};

extern "C" void rover_s3_board_display_state(rover_s3_display_state_t state,
                                               const char *ip, int batt_pct)
{
    bool changed = (state != s_last_state)
                || strcmp(ip ? ip : "", s_last_ip) != 0
                || (batt_pct != s_last_batt);
    if (!changed) return;

    s_last_state = state;
    strlcpy(s_last_ip, ip ? ip : "", sizeof(s_last_ip));
    s_last_batt = batt_pct;

    M5.Display.fillScreen(TFT_BLACK);

    /* State badge */
    M5.Display.setTextSize(2);
    M5.Display.setTextColor(STATE_COLORS[state], TFT_BLACK);
    M5.Display.setCursor(4, 4);
    M5.Display.print(STATE_LABELS[state]);

    /* IP + battery */
    M5.Display.setTextSize(1);
    M5.Display.setTextColor(TFT_WHITE, TFT_BLACK);
    M5.Display.setCursor(4, 40);
    M5.Display.printf("IP: %s", s_last_ip[0] ? s_last_ip : "--");
    M5.Display.setCursor(4, 56);
    M5.Display.printf("BAT: %d%%", batt_pct);
}
```

```cpp
#include "rover_s3_display.h"
#include "setup_device.h"
#include "esp_netif.h"
#include <string.h>

static rover_s3_display_state_t s_state = ROVER_S3_DISPLAY_BOOT;

void rover_s3_display_set_state(rover_s3_display_state_t state)
{
    s_state = state;
    rover_s3_display_refresh();
}

void rover_s3_display_refresh(void)
{
    char ip[20] = "--";
    esp_netif_t *netif = esp_netif_get_handle_from_ifkey("WIFI_STA_DEF");
    if (netif) {
        esp_netif_ip_info_t info;
        if (esp_netif_get_ip_info(netif, &info) == ESP_OK && info.ip.addr != 0) {
            snprintf(ip, sizeof(ip), IPSTR, IP2STR(&info.ip));
        }
    }
    /* Battery handled by setup_device */
    extern int rover_s3_board_get_battery_pct(void);
    int batt = rover_s3_board_get_battery_pct();
    rover_s3_board_display_state(s_state, ip, batt);
}
```

```c
static esp_err_t init_i2s(void)
{
    i2s_chan_config_t chan_cfg = I2S_CHANNEL_DEFAULT_CONFIG(
        CAP_VOICE_I2S_PORT, I2S_ROLE_MASTER);
    ESP_RETURN_ON_ERROR(i2s_new_channel(&chan_cfg, &s_tx_chan, &s_rx_chan),

    i2s_std_config_t std_cfg = {
        .clk_cfg  = I2S_STD_CLK_DEFAULT_CONFIG(CAP_VOICE_SAMPLE_RATE_REC),

            .bclk = CAP_VOICE_I2S_BCLK,

            .dout = CAP_VOICE_I2S_DIN,
            .din  = CAP_VOICE_I2S_DOUT,
        },
    };
    ESP_RETURN_ON_ERROR(i2s_channel_init_std_mode(s_tx_chan, &std_cfg),
                        TAG, "i2s tx init failed");
    ESP_RETURN_ON_ERROR(i2s_channel_init_std_mode(s_rx_chan, &std_cfg),
                        TAG, "i2s rx init failed");
    ESP_RETURN_ON_ERROR(i2s_channel_enable(s_rx_chan), TAG, "i2s rx enable failed");
    return ESP_OK;

static esp_err_t init_codec(void)
{
    const audio_codec_i2c_cfg_t i2c_cfg = {
        .port = CAP_VOICE_I2C_PORT,
        .addr = CAP_VOICE_ES8311_ADDR,
    };
    const audio_codec_data_if_t *i2c_if = audio_codec_new_i2c_data(&i2c_cfg);

    const audio_codec_i2s_cfg_t i2s_cfg = {
        .port = CAP_VOICE_I2S_PORT,
        .rx_handle = s_rx_chan,
        .tx_handle = s_tx_chan,
    };
    const audio_codec_data_if_t *i2s_if = audio_codec_new_i2s_data(&i2s_cfg);

    const es8311_codec_cfg_t es8311_cfg = {0};
    const audio_codec_if_t *es8311_if = es8311_codec_new(&es8311_cfg);

    s_mic_dev = esp_codec_dev_new(&(esp_codec_dev_cfg_t){
        .dev_type = ESP_CODEC_DEV_TYPE_IN,
        .codec_if = es8311_if,
        .data_if  = i2s_if,
    });
    s_spk_dev = esp_codec_dev_new(&(esp_codec_dev_cfg_t){
        .dev_type = ESP_CODEC_DEV_TYPE_OUT,
        .codec_if = es8311_if,
        .data_if  = i2s_if,
    });

    esp_codec_dev_sample_info_t fs = {
        .sample_rate = CAP_VOICE_SAMPLE_RATE_REC,
        .channel     = CAP_VOICE_CHANNELS,
        .bits_per_sample = CAP_VOICE_BITS,
    };
    ESP_RETURN_ON_ERROR(esp_codec_dev_open(s_mic_dev, &fs),
                        TAG, "mic codec open failed");
    esp_codec_dev_set_in_gain(s_mic_dev, 30.0f);  /* dB — tune on hardware */
    ESP_LOGI(TAG, "ES8311 codec init OK");
    return ESP_OK;
}

{
    ESP_RETURN_ON_ERROR(init_i2c(),   TAG, "I2C init failed");
    ESP_RETURN_ON_ERROR(init_i2s(),   TAG, "I2S init failed");
    ESP_RETURN_ON_ERROR(init_codec(), TAG, "codec init failed");
    return ESP_OK;
}

esp_err_t cap_voice_audio_deinit(void)
{
    if (s_mic_dev)  esp_codec_dev_close(s_mic_dev);
    if (s_spk_dev)  esp_codec_dev_close(s_spk_dev);
    if (s_rx_chan)  i2s_channel_disable(s_rx_chan);
    if (s_tx_chan)  i2s_channel_disable(s_tx_chan);
    i2s_del_channel(s_rx_chan);
    i2s_del_channel(s_tx_chan);
    i2c_driver_delete(CAP_VOICE_I2C_PORT);
    return ESP_OK;
}

esp_err_t cap_voice_audio_read(int16_t *buf, size_t samples, size_t *out_read)
{
    size_t bytes_read = 0;
    esp_err_t err = esp_codec_dev_read(s_mic_dev, buf,
                                        samples * sizeof(int16_t));
    if (err == ESP_OK && out_read) {
        *out_read = samples;
    }
    return err;
}

esp_err_t cap_voice_audio_play(const int16_t *buf, size_t samples)
{
    /* Switch sample rate for playback */
    esp_codec_dev_sample_info_t fs = {
        .sample_rate = CAP_VOICE_SAMPLE_RATE_PLAY,
        .channel = CAP_VOICE_CHANNELS,
        .bits_per_sample = CAP_VOICE_BITS,
    };
    esp_codec_dev_close(s_spk_dev);
    esp_codec_dev_open(s_spk_dev, &fs);
    i2s_channel_enable(s_tx_chan);

    esp_err_t err = esp_codec_dev_write(s_spk_dev, (void *)buf,
                                         samples * sizeof(int16_t));

    /* Switch back to record rate */
    esp_codec_dev_close(s_spk_dev);
    fs.sample_rate = CAP_VOICE_SAMPLE_RATE_REC;
    esp_codec_dev_open(s_mic_dev, &fs);
    return err;
}
```

```c
    while (s_running) {

        cap_voice_audio_read(buf, chunk, &got);
        if (got > 0) {
            s_afe_handle->feed(s_afe_data, buf);
        }
        /* Also capture into record_buf when listening */
        if (s_state == CAP_VOICE_STATE_LISTENING && s_record_pos + got < RECORD_BUF_SAMPLES) {
            memcpy(s_record_buf + s_record_pos, buf, got * sizeof(int16_t));
            s_record_pos += got;
        }
    }
    free(buf);
    vTaskDelete(NULL);
}

/* Detect task: reads AFE results, drives state machine */
static void detect_task(void *arg)
{
    static int silence_ms = 0;
    while (s_running) {
        afe_fetch_result_t *res = s_afe_handle->fetch(s_afe_data);
        if (!res) { vTaskDelay(1); continue; }

        if (s_state == CAP_VOICE_STATE_IDLE) {
            if (res->wakeup_state == WAKENET_DETECTED) {
                ESP_LOGI(TAG, "Wake word detected!");
                s_state = CAP_VOICE_STATE_LISTENING;
                s_record_pos = 0;
                silence_ms = 0;
                if (s_cfg.on_wakeup) s_cfg.on_wakeup(s_cfg.cb_ctx);
            }
        } else if (s_state == CAP_VOICE_STATE_LISTENING) {
            /* VAD: count silence */
            if (res->vad_state == AFE_VAD_SILENCE) {
                silence_ms += 32; /* approximate ms per chunk */
            } else {
                silence_ms = 0;
            }
            if (silence_ms >= CONFIG_CAP_VOICE_VAD_SILENCE_MS ||
                    s_record_pos >= RECORD_BUF_SAMPLES) {
                ESP_LOGI(TAG, "Recording done: %u samples", (unsigned)s_record_pos);
                s_state = CAP_VOICE_STATE_RECOGNIZING;
                /* Trigger recognition in a separate task or inline */
                xTaskNotifyGive(s_detect_task); /* signal self to continue */
                /* Recognition happens in next iteration below */
            }
        } else if (s_state == CAP_VOICE_STATE_RECOGNIZING) {
            /* Handled by cap_voice_pipeline_recognize() called from here */
            /* (see Task 12 for MultiNet, Task 13 for Whisper) */
            s_state = CAP_VOICE_STATE_IDLE;
        }
    }
    vTaskDelete(NULL);
}

esp_err_t cap_voice_pipeline_init(const cap_voice_pipeline_config_t *cfg)
{
    s_cfg = *cfg;
    s_record_buf = heap_caps_malloc(RECORD_BUF_SAMPLES * sizeof(int16_t),
                                     MALLOC_CAP_SPIRAM | MALLOC_CAP_8BIT);
    if (!s_record_buf) return ESP_ERR_NO_MEM;

    afe_config_t afe_cfg = AFE_CONFIG_DEFAULT();
    afe_cfg.wakenet_init = true;
    afe_cfg.vad_init = true;
    afe_cfg.memory_alloc_mode = AFE_MEMORY_ALLOC_MORE_PSRAM;

    s_afe_handle = (esp_afe_sr_iface_t *)&ESP_AFE_SR_HANDLE;
    s_afe_data   = s_afe_handle->create_from_config(&afe_cfg);
    if (!s_afe_data) {
        ESP_LOGE(TAG, "AFE create failed");
        return ESP_FAIL;
    }
    ESP_LOGI(TAG, "AFE + WakeNet initialized");
    return ESP_OK;
}

esp_err_t cap_voice_pipeline_start(void)
{
    s_running = true;
    xTaskCreatePinnedToCore(feed_task,   "vfeed",  FEED_TASK_STACK,
                             NULL, PIPELINE_TASK_PRI, &s_feed_task, 1);
    xTaskCreatePinnedToCore(detect_task, "vdetect", DETECT_TASK_STACK,
                             NULL, PIPELINE_TASK_PRI, &s_detect_task, 1);
    ESP_LOGI(TAG, "Pipeline started, listening for wake word");
    return ESP_OK;
}

esp_err_t cap_voice_pipeline_stop(void)
{
    s_running = false;
    return ESP_OK;
}
```

```c
/* Build a minimal WAV header + PCM payload in a single buffer (PSRAM). */
static uint8_t *build_wav(const int16_t *samples, size_t num_samples,
                            size_t *out_len)
{
    const uint32_t sample_rate = 16000;
    const uint16_t channels    = 1;
    const uint16_t bits        = 16;
    const uint32_t data_size   = num_samples * sizeof(int16_t);
    const uint32_t total       = 44 + data_size;

    uint8_t *buf = heap_caps_malloc(total,
                                     MALLOC_CAP_SPIRAM | MALLOC_CAP_8BIT);
    if (!buf) return NULL;

    /* RIFF header */
    memcpy(buf,      "RIFF", 4);
    uint32_t chunk = total - 8;
    memcpy(buf + 4,  &chunk, 4);
    memcpy(buf + 8,  "WAVE", 4);
    memcpy(buf + 12, "fmt ", 4);
    uint32_t fmt_size = 16;
    memcpy(buf + 16, &fmt_size, 4);
    uint16_t audio_fmt = 1;  /* PCM */
    memcpy(buf + 20, &audio_fmt, 2);
    memcpy(buf + 22, &channels, 2);
    memcpy(buf + 24, &sample_rate, 4);
    uint32_t byte_rate = sample_rate * channels * bits / 8;
    memcpy(buf + 28, &byte_rate, 4);
    uint16_t block_align = channels * bits / 8;
    memcpy(buf + 32, &block_align, 2);
    memcpy(buf + 34, &bits, 2);
    memcpy(buf + 36, "data", 4);
    memcpy(buf + 40, &data_size, 4);
    memcpy(buf + 44, samples, data_size);

    *out_len = total;
    return buf;
}

esp_err_t cap_voice_whisper_transcribe(const cap_voice_whisper_config_t *cfg,
                                        const int16_t *samples, size_t num_samples,
                                        char **out_text)
{
    *out_text = NULL;
    size_t wav_len = 0;
    uint8_t *wav = build_wav(samples, num_samples, &wav_len);
    if (!wav) return ESP_ERR_NO_MEM;

    /* Build multipart/form-data body manually */
    const char *boundary  = "----ESP32WavBoundary";
    const char *model     = "whisper-1";
    const char *lang      = cfg->language ? cfg->language : "ru";

    /* Part headers */
    char part_hdr[256];
    int phlen = snprintf(part_hdr, sizeof(part_hdr),
        "--%s\r\nContent-Disposition: form-data; name=\"file\"; filename=\"audio.wav\"\r\n"
        "Content-Type: audio/wav\r\n\r\n", boundary);
    char model_part[256];
    int mplen = snprintf(model_part, sizeof(model_part),
        "\r\n--%s\r\nContent-Disposition: form-data; name=\"model\"\r\n\r\n%s"
        "\r\n--%s\r\nContent-Disposition: form-data; name=\"language\"\r\n\r\n%s"
        "\r\n--%s--\r\n",
        boundary, model, boundary, lang, boundary);

    size_t body_len = phlen + wav_len + mplen;
    uint8_t *body = heap_caps_malloc(body_len,
                                      MALLOC_CAP_SPIRAM | MALLOC_CAP_8BIT);
    if (!body) { free(wav); return ESP_ERR_NO_MEM; }

    memcpy(body,                  part_hdr, phlen);
    memcpy(body + phlen,          wav,      wav_len);
    memcpy(body + phlen + wav_len, model_part, mplen);
    free(wav);

    /* Build URL */
    char url[256];
    snprintf(url, sizeof(url), "%s/v1/audio/transcriptions",
             cfg->base_url ? cfg->base_url : "https://api.openai.com");

    char auth[ROVER_S3_STR_LEN + 16];
    snprintf(auth, sizeof(auth), "Bearer %s", cfg->api_key ? cfg->api_key : "");

    char ct[96];
    snprintf(ct, sizeof(ct), "multipart/form-data; boundary=%s", boundary);

    /* Send */
    esp_http_client_config_t http_cfg = {
        .url = url,
        .crt_bundle_attach = esp_crt_bundle_attach,
        .timeout_ms = 30000,
    };
    esp_http_client_handle_t client = esp_http_client_init(&http_cfg);
    esp_http_client_set_method(client, HTTP_METHOD_POST);
    esp_http_client_set_header(client, "Authorization", auth);
    esp_http_client_set_header(client, "Content-Type", ct);
    esp_http_client_set_post_field(client, (char *)body, (int)body_len);

    esp_err_t err = esp_http_client_perform(client);
    int status = esp_http_client_get_status_code(client);
    ESP_LOGI(TAG, "Whisper HTTP status=%d", status);

    if (err == ESP_OK && status == 200) {
        int content_len = esp_http_client_get_content_length(client);
        char *resp = calloc(1, content_len + 1);
        if (resp) {
            esp_http_client_read_response(client, resp, content_len);
            cJSON *j = cJSON_Parse(resp);
            cJSON *text_j = j ? cJSON_GetObjectItem(j, "text") : NULL;
            if (text_j && cJSON_IsString(text_j)) {
                *out_text = strdup(text_j->valuestring);
                ESP_LOGI(TAG, "Transcript: %s", *out_text);
            }
            cJSON_Delete(j);
            free(resp);
        }
    } else {
        err = ESP_FAIL;
    }

    esp_http_client_cleanup(client);
    free(body);
    return (*out_text) ? ESP_OK : ESP_FAIL;
}
```

```c
/* Dynamic response buffer grown on each HTTP event */
typedef struct {

    size_t   len;
    size_t   cap;
} tts_buf_t;

static esp_err_t on_data(esp_http_client_event_t *evt)
{
    if (evt->event_id != HTTP_EVENT_ON_DATA) return ESP_OK;
    tts_buf_t *b = (tts_buf_t *)evt->user_data;
    if (b->len + evt->data_len > b->cap) {
        size_t new_cap = b->cap + evt->data_len + 65536;
        uint8_t *tmp = heap_caps_realloc(b->data, new_cap,
                                          MALLOC_CAP_SPIRAM | MALLOC_CAP_8BIT);
        if (!tmp) return ESP_ERR_NO_MEM;
        b->data = tmp;
        b->cap  = new_cap;
    }
    memcpy(b->data + b->len, evt->data, evt->data_len);
    b->len += evt->data_len;
    return ESP_OK;
}

        cfg->voice ? cfg->voice : "alloy");

    char *body_str = cJSON_PrintUnformatted(body);
    cJSON_Delete(body);

        .len  = 0,
        .cap  = 65536,
    };
    if (!buf.data) { free(body_str); return ESP_ERR_NO_MEM; }

    esp_http_client_config_t http_cfg = {
```

```c
}

/* LLM tool: voice_set_voice */
static claw_cap_result_t voice_set_voice_execute(const claw_cap_call_t *call)
{
    const cJSON *args = cJSON_Parse(call->arguments_json);
    const char *voice = cJSON_GetStringValue(cJSON_GetObjectItem(args, "voice"));
    if (voice) {
        strlcpy((char *)s_config.tts_voice, voice, 32);
    }
    cJSON_Delete((cJSON *)args);
    return claw_cap_result_ok("{\"ok\":true}");
}

        .id          = "voice_say",
        .name        = "voice_say",
        .family      = "voice",

        .cap_flags   = CLAW_CAP_FLAG_CALLABLE_BY_LLM,
        .input_schema_json =

        .cap_flags   = CLAW_CAP_FLAG_CALLABLE_BY_LLM,
        .input_schema_json =
            "{\"type\":\"object\",\"properties\":{\"voice\":{\"type\":\"string\","

            "\"required\":[\"voice\"]}",
        .execute     = voice_set_voice_execute,
    },
};

static const claw_cap_group_t s_voice_group = {
    .group_id        = "cap_voice",
    .descriptors     = s_voice_descriptors,
    .descriptor_count = sizeof(s_voice_descriptors) / sizeof(s_voice_descriptors[0]),
};

esp_err_t cap_voice_register_group(void)
{
    return claw_cap_register_group(&s_voice_group);
}

esp_err_t cap_voice_set_config(const cap_voice_config_t *config)
{
    s_config = *config;
    return ESP_OK;
}

esp_err_t cap_voice_start(void)
{
    ESP_RETURN_ON_ERROR(cap_voice_audio_init(), TAG, "audio init failed");

    cap_voice_pipeline_config_t pcfg = {
        .on_wakeup       = on_wakeup,
        .on_command      = on_command,
        .on_transcript   = on_transcript,
        .cb_ctx          = NULL,
        .wake_sensitivity    = s_config.wake_sensitivity > 0
                               ? s_config.wake_sensitivity : 0.7f,
        .multinet_threshold  = s_config.multinet_threshold > 0
                               ? s_config.multinet_threshold : 0.85f,
        .whisper_api_key = s_config.whisper_api_key,
        .whisper_base_url = s_config.whisper_base_url,
        .tts_api_key     = s_config.tts_api_key,
        .tts_base_url    = s_config.tts_base_url,
    };
    ESP_RETURN_ON_ERROR(cap_voice_pipeline_init(&pcfg), TAG, "pipeline init failed");
    return cap_voice_pipeline_start();
}

esp_err_t cap_voice_stop(void)
{
    cap_voice_pipeline_stop();
    return cap_voice_audio_deinit();
}

esp_err_t cap_voice_speak(const char *text)
{
    cap_voice_tts_config_t tcfg = {
        .api_key  = s_config.tts_api_key,
        .base_url = s_config.tts_base_url,
        .voice    = s_config.tts_voice,
    };
    return cap_voice_tts_speak(&tcfg, text);
}
```
